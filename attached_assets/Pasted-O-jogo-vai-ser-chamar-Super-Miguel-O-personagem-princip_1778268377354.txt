// ======================================
// UPDATE
// ======================================

function update(){

    player.setVelocityX(0);

    // ANDAR ESQUERDA
    if(cursors.left.isDown){

        player.setVelocityX(-220);

        player.flipX = true;
    }

    // ANDAR DIREITA
    if(cursors.right.isDown){

        player.setVelocityX(220);

        player.flipX = false;
    }

    // CORRER
    if(cursors.shift.isDown){

        if(cursors.right.isDown){

            player.setVelocityX(420);
        }

        if(cursors.left.isDown){

            player.setVelocityX(-420);
        }
    }

    // PULAR
    if(cursors.up.isDown && player.body.touching.down){

        player.setVelocityY(-500);
    }

    // ABAIXAR
    if(cursors.down.isDown){

        player.setScale(0.8);

    }else{

        player.setScale(1);
    }

    atualizarHUD();
}

function atualizarHUD(){

    pontosTexto.setText(
        "Pontos: " + pontos +
        " | Energia: " + energia +
        " | Fase: " + fase
    );

    if(energia <= 0){

        missaoTexto.setText(
            "Você ficou cansado! Vá dormir cedo!"
        );
    }

    // AVANÇAR FASES
    if(pontos >= 50 && fase == 1){

        fase = 2;

        missaoTexto.setText(
            "FASE 2: Hora de brincar no quintal!"
        );
    }

    if(pontos >= 120 && fase == 2){

        fase = 3;

        missaoTexto.setText(
            "FASE 3: Passeio no parque!"
        );
    }

    if(pontos >= 200 && fase == 3){

        fase = 4;

        missaoTexto.setText(
            "FASE 4: Hora de jantar e escovar os dentes!"
        );
    }

    if(pontos >= 300){

        fase = 5;

        missaoTexto.setText(
            "Parabéns! Agora vá dormir cedo!"
        );
    }
}

class CidadeScene extends Phaser.Scene {

    constructor(){
        super("CidadeScene");
    }

    preload(){

        this.load.image('tiles','assets/cidade/tiles.png');

        this.load.tilemapTiledJSON(
            'mapa',
            'assets/cidade/mapa.json'
        );

    }

    create(){

        const mapa = this.make.tilemap({
            key:'mapa'
        });

        const tileset = mapa.addTilesetImage(
            'tiles',
            'tiles'
        );

        const chao = mapa.createLayer(
            'Chao',
            tileset
        );

        const ruas = mapa.createLayer(
            'Ruas',
            tileset
        );

        const casas = mapa.createLayer(
            'Casas',
            tileset
        );

        ruas.setCollisionByProperty({
            collider:true
        });

        player = this.physics.add.sprite(
            400,
            300,
            'player'
        );

        this.physics.add.collider(
            player,
            ruas
        );

        this.cameras.main.startFollow(player);

        this.cameras.main.setZoom(2);

    }
}

PASSO 7 — CENA DA CIDADE
LagoaDouradaScene.js
class LagoaDouradaScene extends Phaser.Scene {

    constructor(){

        super("LagoaDouradaScene");
    }

    preload(){

        this.load.image(
            'tiles',
            'assets/tiles/cidade.png'
        );

        this.load.tilemapTiledJSON(
            'mapa',
            'assets/mapas/lagoa.json'
        );

        this.load.spritesheet(
            'player',
            'assets/personagens/player.png',
            {
                frameWidth:32,
                frameHeight:32
            }
        );
    }

    create(){

        const mapa = this.make.tilemap({
            key:'mapa'
        });

        const tileset = mapa.addTilesetImage(
            'cidade',
            'tiles'
        );

        const chao = mapa.createLayer(
            'Ground',
            tileset
        );

        const ruas = mapa.createLayer(
            'Roads',
            tileset
        );

        const casas = mapa.createLayer(
            'Houses',
            tileset
        );

        const colisao = mapa.createLayer(
            'Collision',
            tileset
        );

        colisao.setCollisionByProperty({
            collider:true
        });

        player = this.physics.add.sprite(
            300,
            300,
            'player'
        );

        this.physics.add.collider(
            player,
            colisao
        );

        this.cameras.main.startFollow(player);

    }

    update(){

        movimentarPlayer();
    }
}
PASSO 8 — MOVIMENTO
function movimentarPlayer(){

    player.setVelocity(0);

    if(cursors.left.isDown){

        player.setVelocityX(-200);
    }

    if(cursors.right.isDown){

        player.setVelocityX(200);
    }

    if(cursors.up.isDown){

        player.setVelocityY(-200);
    }

    if(cursors.down.isDown){

        player.setVelocityY(200);
    }
}
